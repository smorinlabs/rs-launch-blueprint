#!/bin/zsh
# Serial Codex RESEARCH runner, modelled on audit-runner.sh: re-reads the jobs
# file each cycle, waits while any codex exec runs (the anonymous GitHub budget
# is per-IP and shared), runs codex-research.sh for the first job without a
# marker, and marks it done ONLY when raw/codex.md exists, exit is 0 and the
# receipt names it. Bounded to 8 hours.
set -u
SP=/private/tmp/claude-501/-Users-stevemorin-c-rs-launch-blueprint-p02-plan/b7c06b2f-d78d-4987-b237-63eeced4bd23/scratchpad
REPO=/Users/stevemorin/c/rs-launch-blueprint-p02-plan
JOBS=$SP/research-jobs.tsv
DONE=$SP/research-done.txt
FAILED=$SP/research-failed.txt
touch "$DONE" "$FAILED"
[[ -f $SP/codex-research.sh ]] || { echo "FATAL: codex-research.sh missing"; exit 2; }
for cycle in {1..960}; do
  if pgrep -f "codex exec --skip-git-repo-check" > /dev/null 2>&1; then sleep 30; continue; fi
  picked=""
  while IFS=$'\t' read -r ITEM RUN KIND; do
    [[ -z "${ITEM:-}" ]] && continue
    grep -Fxq "$ITEM $RUN" "$DONE" && continue
    grep -Fxq "$ITEM $RUN" "$FAILED" && continue
    picked="yes"
    echo "$(date -u +%H:%M:%SZ) START $ITEM $RUN ($KIND)"
    zsh $SP/codex-research.sh "$ITEM" "$RUN" "$KIND" > $SP/$ITEM-research.out 2>&1
    rc=$?
    d=$REPO/research/runs/$ITEM/$RUN
    ex=$(cat "$d/raw/codex-exit.txt" 2>/dev/null)
    if [[ $rc -eq 0 && "$ex" == "0" && -s "$d/raw/codex.md" && -s "$d/raw/codex-receipt.txt" ]]; then
      echo "$ITEM $RUN" >> "$DONE"; echo "$(date -u +%H:%M:%SZ) DONE  $ITEM $RUN bytes=$(wc -c < "$d/raw/codex.md")"
    else
      echo "$ITEM $RUN" >> "$FAILED"; echo "$(date -u +%H:%M:%SZ) FAIL  $ITEM $RUN rc=$rc exit='$ex' -- see $SP/$ITEM-research.out"
    fi
    break
  done < "$JOBS"
  [[ -z "$picked" ]] && sleep 30
done
echo "$(date -u +%H:%M:%SZ) research runner window ended"
