#!/bin/zsh
# usage: doxa-status-line.sh <op> — one line: op-suffix status provider=state ...
cd /Users/stevemorin/c/rs-launch-blueprint-p02-plan || exit 9

    d=json.load(sys.stdin)["data"]; p=d.get("providers",{})

    print(sys.argv[1][-16:], "status-error", type(e).__name__)
' "$1"
