#!/usr/bin/env python
"""Read usage metadata back from each provider's API for one collected Doxa op.

usage: doxa-usage-readback.py <run-dir> <openai_resp_id> <perplexity_req_id> <gemini_interaction_id> [--out <path>]

Run with the Doxa virtualenv python (needs openai, google-genai, httpx).
Read-only: one GET per provider.  Writes operations/doxa-usage.json in the shape
Wave 0 used (see research/runs/R69/.../operations/doxa-usage.json).
"""
import asyncio
import collections
import json
import os
import pathlib
import sys

OPENAI_IN = 4.0 / 1_000_000       # USD per input token, list
OPENAI_OUT = 20.0 / 1_000_000     # USD per output token, list
OPENAI_SEARCH = 10.0 / 1_000      # USD per web_search call, list


def openai_usage(resp_id):
    import httpx
    key = os.environ["OPENAI_API_KEY"]
    r = httpx.get(
        f"https://api.openai.com/v1/responses/{resp_id}",
        headers={"Authorization": f"Bearer {key}"},
        timeout=120.0,
    )
    r.raise_for_status()
    d = r.json()
    types = collections.Counter(item.get("type") for item in d.get("output", []))
    usage = d.get("usage") or {}
    n_search = types.get("web_search_call", 0)
    est = (
        usage.get("input_tokens", 0) * OPENAI_IN
        + usage.get("output_tokens", 0) * OPENAI_OUT
        + n_search * OPENAI_SEARCH
    )
    return {
        "status": d.get("status"),
        "model": d.get("model"),
        "created_at": d.get("created_at"),
        "completed_at": d.get("completed_at"),
        "usage": usage,
        "reasoning": d.get("reasoning"),
        "output_item_types": dict(types),
        "list_price_usd_estimate": round(est, 4),
    }


def perplexity_usage(req_id):
    import httpx
    key = os.environ["PERPLEXITY_API_KEY"]
    r = httpx.get(
        f"https://api.perplexity.ai/async/chat/completions/{req_id}",
        headers={"Authorization": f"Bearer {key}"},
        timeout=120.0,
    )
    r.raise_for_status()
    d = r.json()
    resp = d.get("response") or {}
    usage = resp.get("usage") or {}
    choices = resp.get("choices") or [{}]
    cites = resp.get("citations") or resp.get("search_results") or []
    return {
        "status": d.get("status"),
        "model": resp.get("model") or d.get("model"),
        "created_at": d.get("created_at"),
        "started_at": d.get("started_at"),
        "completed_at": d.get("completed_at"),
        "usage": usage,
        "n_citations": len(cites),
        "finish_reason": choices[0].get("finish_reason"),
        "list_price_usd_estimate": (usage.get("cost") or {}).get("total_cost"),
    }


def gemini_usage(interaction_id):
    from google import genai
    client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])
    inter = asyncio.run(client.aio.interactions.get(id=interaction_id))
    d = inter.model_dump(mode="json", exclude_none=True)
    um = d.get("usage") or d.get("usage_metadata") or d.get("usageMetadata") or {}
    return {
        "status": d.get("status") or d.get("state"),
        "agent": d.get("agent") or d.get("model"),
        "created": d.get("created") or d.get("create_time"),
        "updated": d.get("updated") or d.get("update_time"),
        "n_steps": len(d.get("steps") or []),
        "usage": um,
        "list_price_usd_estimate": "flat USD 1-6 (Gemini Deep Research is not "
                                   "priced per token in the published list)",
    }


def main():
    args = sys.argv[1:]
    run_dir = pathlib.Path(args[0])
    oa, px, gm = args[1], args[2], args[3]
    out = None
    for i, a in enumerate(args):
        if a == "--out":
            out = args[i + 1]
    result = {}
    for name, fn, arg in (("openai", openai_usage, oa),
                          ("perplexity", perplexity_usage, px),
                          ("gemini", gemini_usage, gm)):
        try:
            result[name] = fn(arg)
        except Exception as exc:  # recorded, never silently dropped
            result[name] = {"read_back_error": f"{type(exc).__name__}: {exc}"}
    path = pathlib.Path(out) if out else run_dir / "operations" / "doxa-usage.json"
    path.write_text(json.dumps(result, indent=1, sort_keys=False) + "\n")
    print(f"wrote {path}")
    print(json.dumps(result, indent=1)[:4000])


main()
