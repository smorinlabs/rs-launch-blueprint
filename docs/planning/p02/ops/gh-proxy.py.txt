#!/usr/bin/env python3
"""Local authenticated proxy for api.github.com on 127.0.0.1:8765.
The token is read once from `gh auth token` and injected outside the sandbox;
clients send no credential and never see one. GET/HEAD only. Logs paths, never headers."""
import http.server, subprocess, sys, urllib.request, urllib.error, socketserver
TOKEN = subprocess.run(["gh", "auth", "token"], capture_output=True, text=True, check=True).stdout.strip()
UPSTREAM = "https://api.github.com"
class H(http.server.BaseHTTPRequestHandler):
    def _go(self, method):
        if method not in ("GET", "HEAD"):
            self.send_response(405); self.end_headers(); return
        url = UPSTREAM + self.path
        req = urllib.request.Request(url, method=method, headers={
            "Authorization": "Bearer " + TOKEN, "Accept": self.headers.get("Accept", "application/vnd.github+json"),
            "User-Agent": "rs-launch-blueprint-research-proxy", "X-GitHub-Api-Version": "2022-11-28"})
        try:
            with urllib.request.urlopen(req, timeout=60) as r:
                body = r.read(); self.send_response(r.status)
                for k in ("Content-Type", "X-RateLimit-Limit", "X-RateLimit-Remaining", "X-RateLimit-Reset", "Link", "ETag"):
                    v = r.headers.get(k)
                    if v: self.send_header(k, v)
                self.send_header("Content-Length", str(len(body))); self.end_headers()
                if method == "GET": self.wfile.write(body)
        except urllib.error.HTTPError as e:
            body = e.read(); self.send_response(e.code)
            for k in ("Content-Type", "X-RateLimit-Limit", "X-RateLimit-Remaining", "X-RateLimit-Reset"):
                v = e.headers.get(k)
                if v: self.send_header(k, v)
            self.send_header("Content-Length", str(len(body))); self.end_headers()
            if method == "GET": self.wfile.write(body)
        except Exception as e:
            msg = ("proxy error: " + type(e).__name__).encode(); self.send_response(502)
            self.send_header("Content-Length", str(len(msg))); self.end_headers(); self.wfile.write(msg)
    def do_GET(self): self._go("GET")
    def do_HEAD(self): self._go("HEAD")
    def log_message(self, fmt, *args): sys.stderr.write("%s %s\n" % (self.command, self.path.split("?")[0]))
class S(socketserver.ThreadingMixIn, http.server.HTTPServer): daemon_threads = True
if __name__ == "__main__":
    S(("127.0.0.1", 8765), H).serve_forever()
